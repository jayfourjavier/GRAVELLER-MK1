    void handleCommand() {
        String action = server.arg("action");
        String value = server.arg("value");
        
        // ONLY update the command variable - let main loop handle everything
        if (action == "forward") {
            Serial.println("WEB: FORWARD");
            command = CMD_MOVE_FORWARD;
        }
        else if (action == "backward") {
            Serial.println("WEB: BACKWARD");
            command = CMD_MOVE_BACKWARD;
        }
        else if (action == "left") {
            Serial.println("WEB: LEFT");
            command = CMD_MOVE_LEFT;
        }
        else if (action == "right") {
            Serial.println("WEB: RIGHT");
            command = CMD_MOVE_RIGHT;
        }
        else if (action == "drop") {
            Serial.println("WEB: DROP");
            command = CMD_CONTAINER_ON;
        }
        else if (action == "mode") {
            Serial.println("WEB: MODE TOGGLE");
            command = CMD_MODE_DROP;  // Will toggle in main loop
        }
        else if (action == "camera") {
            Serial.println("WEB: CAMERA TOGGLE");
            command = CMD_FRONT_CAM;  // Will toggle in main loop
        }
        else if (action == "lamp") {
            Serial.println("WEB: LAMP TOGGLE");
            command = CMD_LAMP_ON;  // Will toggle in main loop
        }
        else if (action == "camerapower") {
            if (value == "on") {
                Serial.println("WEB: CAMERA ON");
                command = CMD_CAMERA_ON;
            } else {
                Serial.println("WEB: CAMERA OFF");
                command = CMD_CAMERA_OFF;
            }
        }
        else if (action == "speed") {
            ContainerSpeed = value.toInt();
            Serial.printf("WEB: SPEED SET TO %d%%\n", ContainerSpeed);
        }
        
        // Send current status back
        JsonDocument doc;
        doc["speed"] = ContainerSpeed;
        doc["mode"] = (containerMode == 0) ? "DROP" : "LOAD";  // Assuming 0=DROP, 1=LOAD
        doc["camera"] = (command == CMD_FRONT_CAM) ? "FRONT" : "BACK";
        doc["lamp"] = "OFF";  // Will be updated by main loop
        
        String response;
        serializeJson(doc, response);
        server.send(200, "application/json", response);
        
        // Reset command after a short delay for movement commands
        if (action == "forward" || action == "backward" || action == "left" || action == "right") {
            delay(300);
            command = CMD_STOP;
        } else if (action == "drop") {
            delay(500);
            command = CMD_CONTAINER_OFF;
        }
    }
    
    void handleStatus() {
        JsonDocument doc;
        doc["speed"] = ContainerSpeed;
        doc["mode"] = (containerMode == 0) ? "DROP" : "LOAD";
        doc["camera"] = (command == CMD_FRONT_CAM) ? "FRONT" : "BACK";
        doc["lamp"] = "OFF";
        
        String response;
        serializeJson(doc, response);
        server.send(200, "application/json", response);
    }
